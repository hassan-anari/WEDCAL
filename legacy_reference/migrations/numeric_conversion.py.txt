def floor_decimal(value):
    if value is None:
        return 0
    return int(Decimal(str(value)).to_integral_value(rounding=ROUND_FLOOR))


def referral_parts(value):
    if value is None:
        return 0, 0
    hundredths = int((Decimal(str(value)) * Decimal('100')).to_integral_value(rounding=ROUND_FLOOR))
    return hundredths // 100, hundredths % 100
