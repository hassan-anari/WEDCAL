    def test_self_unknown_malformed_and_sentinel_are_ignored(self):
        for value in [self.referrer.referral_code, "unknownunknown22", "INVALID!", START_REFERRAL_SENTINEL]:
            with self.subTest(value=value):
                target_id = self.referrer.telegram_id if value == self.referrer.referral_code else 9100 + len(str(value))
                if target_id != self.referrer.telegram_id:
                    TelegramProfile.objects.filter(telegram_id=target_id).delete()
                result = assign_referrer_from_code(invitee_telegram_id=target_id, referral_value=value)
                profile = TelegramProfile.objects.get(telegram_id=target_id)
                self.assertFalse(result.assigned)
                if target_id == self.referrer.telegram_id:
                    self.assertIsNone(profile.referred_by)
