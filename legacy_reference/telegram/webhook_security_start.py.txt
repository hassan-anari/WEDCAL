def normalize_referral_code(value: str | None) -> str | None:
    if value is None:
        return None
    value = str(value).strip().lower()
    if not value:
        return None
    return value if REFERRAL_CODE_PATTERN.fullmatch(value) else None

def parse_start_referral_value(text: str | None) -> tuple[bool, str]:
    if not text:
        return False, START_REFERRAL_SENTINEL
    parts = text.strip().split()
    if not parts:
        return False, START_REFERRAL_SENTINEL

    command = parts[0]
    if command == "/start":
        pass
    elif command.startswith("/start@"):
        suffix = command[len("/start@"):]
        if not suffix or not re.fullmatch(r"[A-Za-z0-9_]{1,64}", suffix):
            return False, START_REFERRAL_SENTINEL
    else:
        return False, START_REFERRAL_SENTINEL

    if len(parts) == 1:
        return True, START_REFERRAL_SENTINEL
    if len(parts) != 2:
        return True, START_REFERRAL_SENTINEL

    candidate = parts[1].strip()
    if len(candidate) > 64:
        return True, START_REFERRAL_SENTINEL
    normalized = normalize_referral_code(candidate)
    return True, normalized or START_REFERRAL_SENTINEL

def validate_telegram_webhook_secret(
    x_telegram_bot_api_secret_token: str | None = Header(
        default=None, alias="X-Telegram-Bot-Api-Secret-Token"
    ),
) -> None:
    expected_secret = get_telegram_secret_token()
    if not expected_secret:
        raise HTTPException(status_code=503, detail="Webhook authentication unavailable")
    if not x_telegram_bot_api_secret_token or not secrets.compare_digest(
        x_telegram_bot_api_secret_token, expected_secret
    ):
        raise HTTPException(status_code=403, detail="Forbidden")
