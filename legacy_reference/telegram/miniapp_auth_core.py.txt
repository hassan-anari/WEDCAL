    def _get_bot_token(self):
        try:
            bot_token = config('TELEGRAM_BOT_TOKEN')
        except UndefinedValueError:
            logger.error("Telegram bot token is not configured", extra={"auth_stage": "configuration"})
            raise AuthenticationFailed("Server configuration error.")

        if not bot_token:
            logger.error("Telegram bot token is empty", extra={"auth_stage": "configuration"})
            raise AuthenticationFailed("Server configuration error.")
        return bot_token

    def _validate_auth_date(self, auth_date):
        if not auth_date:
            raise AuthenticationFailed("Missing auth_date in initData.")

        try:
            auth_timestamp = int(auth_date)
        except (TypeError, ValueError):
            raise AuthenticationFailed("Malformed auth_date in initData.")

        current_timestamp = int(time.time())
        if current_timestamp - auth_timestamp > INIT_DATA_MAX_AGE_SECONDS:
            raise AuthenticationFailed("Token has expired.")
        if auth_timestamp - current_timestamp > INIT_DATA_FUTURE_SKEW_SECONDS:
            raise AuthenticationFailed("auth_date is too far in the future.")
        return auth_timestamp

    def _parse_user_data(self, user_data_str):
        if not user_data_str:
            raise AuthenticationFailed("Missing user data in initData.")

        try:
            user_data = json.loads(user_data_str)
        except json.JSONDecodeError:
            raise AuthenticationFailed("Invalid user JSON in initData.")
