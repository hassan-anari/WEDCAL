ALBUM_ACCUMULATOR_SCRIPT = """
local meta_key = KEYS[1]
local messages_key = KEYS[2]
local finalize_key = KEYS[3]
local metadata = ARGV[1]
local expected_admin_id = ARGV[2]
local expected_source_chat_id = ARGV[3]
local message_id = tonumber(ARGV[4])
local ttl = tonumber(ARGV[5])

local existing = redis.call('GET', meta_key)
if existing then
  local decoded = cjson.decode(existing)
  if tostring(decoded['admin_user_id']) ~= expected_admin_id or tostring(decoded['source_chat_id']) ~= expected_source_chat_id then
    return {err='album metadata mismatch'}
  end
else
  redis.call('SET', meta_key, metadata, 'EX', ttl)
end
redis.call('ZADD', messages_key, 'NX', message_id, message_id)
redis.call('EXPIRE', meta_key, ttl)
redis.call('EXPIRE', messages_key, ttl)
local scheduled = redis.call('SET', finalize_key, '1', 'EX', ttl, 'NX')
if scheduled then
  return {1, redis.call('ZCARD', messages_key)}
end
redis.call('EXPIRE', finalize_key, ttl)
return {0, redis.call('ZCARD', messages_key)}
"""

async def collect_album_message_atomic(*, admin_user_id: int, media_group_id: str, audience: str, source_chat_id: int, message_id: int) -> tuple[bool, int]:
    metadata = json.dumps({
        "audience": audience,

        "source_chat_id": source_chat_id,

        str(source_chat_id),
        int(message_id),
        BROADCAST_ALBUM_TTL_SECONDS,
    )
    return bool(int(scheduled)), int(count)

def parse_broadcast_state(raw_state: str | None) -> dict[str, Any] | None:
    if not raw_state:
        return None
    try:
        state = json.loads(raw_state)
    except (TypeError, ValueError):
        return None
    if not isinstance(state, dict):
        return None
    if state.get("audience") not in VALID_BROADCAST_AUDIENCES:
        return None
    if state.get("origin_chat_type") != "private":
        return None
    try:
        state["admin_user_id"] = int(state["admin_user_id"])
        state["origin_chat_id"] = int(state["origin_chat_id"])
    except (KeyError, TypeError, ValueError):
        return None
    return state

async def queue_admin_broadcast(
    *, target_audience: str, source_chat_id: int, source_message_ids: list[int], admin_chat_id: int,
    update_id: int | None, album: bool = False, state_key: str | None = None, media_group_id: str | None = None,
) -> bool:
    try:
        result = celery_app.send_task(
            "apps.telegram_airdrop.tasks.admin_broadcast",
            kwargs={
                "target_audience": target_audience,
                "source_chat_id": source_chat_id,
                "source_message_ids": source_message_ids,
            },
        )
    except Exception:
        if state_key:
            await redis_client.delete(state_key)
        if media_group_id:
            await redis_client.delete(broadcast_album_meta_key(admin_chat_id, media_group_id), broadcast_album_messages_key(admin_chat_id, media_group_id), broadcast_album_finalize_key(admin_chat_id, media_group_id))
        await clear_admin_album_keys(admin_chat_id)
        logger.exception("admin_broadcast_queue_failure", extra={"event": "admin_broadcast_queue_failure", "audience": target_audience, "admin_user_id": admin_chat_id, "source_message_count": len(source_message_ids)})
        await send_telegram("sendMessage", {"chat_id": admin_chat_id, "text": QUEUE_FAILURE_MESSAGE}, update_id=update_id, action="admin_broadcast_queue_failure")
        return False
    if state_key:
        await redis_client.delete(state_key)
    if media_group_id:
        await redis_client.delete(broadcast_album_meta_key(admin_chat_id, media_group_id), broadcast_album_messages_key(admin_chat_id, media_group_id), broadcast_album_finalize_key(admin_chat_id, media_group_id))
    logger.info("admin_broadcast_queue_success", extra={"event": "admin_broadcast_queue_success", "audience": target_audience, "admin_user_id": admin_chat_id, "source_message_count": len(source_message_ids), "celery_task_id": getattr(result, "id", None)})
    text = f"{'Media album broadcast' if album else 'Broadcast'} to '{target_audience}' queued successfully."
    await send_telegram("sendMessage", {"chat_id": admin_chat_id, "text": text}, update_id=update_id, action="admin_broadcast_queued")
    return True
