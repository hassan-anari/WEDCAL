    def test_no_active_season_multiple_active_seasons_and_zero_reward_fail_closed(self):
        cases = [
            ([], "No active airdrop season is available."),
            ([0], "Hourly claim reward is not configured for the active season."),
            ([100, 200], "Airdrop season configuration is invalid."),
        ]
        for rewards, message in cases:
            with self.subTest(message=message):
                AirdropSeason.objects.all().delete()
                TelegramProfile.objects.all().delete()
                UserActivityLog.objects.all().delete()
                for idx, reward in enumerate(rewards):
                    AirdropSeason.objects.create(title=f"Season {idx}", is_active=True, hourly_claim_reward=reward)

                response = self._post_claim()

                self.assertEqual(response.status_code, 400)
                self.assertEqual(response.json()["message"], message)
                profile = self._profile()
                self.assertEqual(profile.bonus_points, 0)
                self.assertIsNone(profile.last_claim_at)
                self.assertFalse(UserActivityLog.objects.filter(user=profile).exists())

    def test_successful_retry_is_idempotent_and_current_season_only(self):
        season = self.expired_season()
        profile = TelegramProfile.objects.create(telegram_id=1, total_archived_points=1000, activity_earned_points=3, group_earned_points=4, referral_earned_points=2, bonus_points=1)

                check_and_transition_season()

        record = UserSeasonRecord.objects.get(user=profile, season=season)

        self.assertEqual(profile.activity_earned_points, 0)

    def test_second_season_records_only_second_season_earnings(self):
        first = self.expired_season("First")
        profile = TelegramProfile.objects.create(telegram_id=1, activity_earned_points=10)
        check_and_transition_season()
        second = self.expired_season("Second")
        profile.refresh_from_db()
        profile.activity_earned_points = 7
        profile.group_earned_points = 3
        profile.save(update_fields=["activity_earned_points", "group_earned_points"])
        check_and_transition_season()
        self.assertEqual(UserSeasonRecord.objects.get(user=profile, season=first).total_points, 10)
        self.assertEqual(UserSeasonRecord.objects.get(user=profile, season=second).total_points, 10)
        profile.refresh_from_db()
        self.assertEqual(profile.total_archived_points, 20)
