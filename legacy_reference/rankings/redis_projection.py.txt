def _mark_dirty_if_rebuilding(client, profile_id: int) -> None:
    try:
        if client.exists(LEADERBOARD_REBUILD_LOCK_KEY):
            client.sadd(LEADERBOARD_REBUILD_DIRTY_KEY, str(profile_id))
            client.expire(LEADERBOARD_REBUILD_DIRTY_KEY, REBUILD_DIRTY_TTL_SECONDS)
    except Exception:
        logger.warning("Unable to mark leaderboard dirty profile", exc_info=True, extra={"profile_id": profile_id})

def sync_profiles(profile_ids: Iterable[int], *, redis_client=None, mark_dirty: bool = True) -> dict[str, int]:
    ids = sorted({int(profile_id) for profile_id in profile_ids if profile_id})
    result = {"requested": len(ids), "synced": 0, "failed": 0}
    if not ids:
        return result
    client = _client(redis_client)
    for profile_id in ids:
        if sync_profile(profile_id, redis_client=client, mark_dirty=mark_dirty):
            result["synced"] += 1
        else:
            result["failed"] += 1
    return result

def get_competition_rank(telegram_id: int | str, *, redis_client=None) -> int | None:
    client = _client(redis_client)
    member = leaderboard_member(telegram_id)
    try:
        score = client.zscore(LEADERBOARD_KEY, member)
        if score is None:
            return None
        score = coerce_leaderboard_score(score)
        greater_count = client.zcount(LEADERBOARD_KEY, f"({score}", "+inf")
        return int(greater_count) + 1
    except Exception:
        logger.warning("Unable to read Telegram leaderboard rank", exc_info=True, extra={"telegram_id": str(telegram_id)})
        return None

# --- _parse_redis_member (lines 199-206) ---
def _parse_redis_member(member: Any) -> int | None:
    try:
        text = member.decode() if isinstance(member, bytes) else str(member)
        if not text.lstrip("-").isdigit():
            return None
        return int(text)
    except Exception:
        return None

def _score_groups_desc(client, *, page_size: int = 50):
    seen = set()
    max_score = "+inf"
    page_size = max(int(page_size), 1)
    while True:
        raw_scores = client.zrevrangebyscore(
            LEADERBOARD_KEY,
            max_score,
            "-inf",
            start=0,
            num=page_size,
            withscores=True,
        )
        if not raw_scores:
            return

        next_max_score = None
        for _, raw_score in raw_scores:
            try:
                score = coerce_leaderboard_score(raw_score)
            except Exception:
                continue
            next_max_score = score
            if score in seen:
                continue
            seen.add(score)
            yield score

        if next_max_score is None or len(raw_scores) < page_size:
            return
        max_score = f"({next_max_score}"

def get_top_rows(limit: int = DEFAULT_TOP_LIMIT, *, redis_client=None, score_page_size: int = 50) -> list[dict[str, Any]]:
    client = _client(redis_client)
    rows: list[dict[str, Any]] = []
    malformed_members: list[str] = []
    stale_members: list[str] = []
    try:
        for boundary_score in _score_groups_desc(client, page_size=score_page_size):
            raw_members = client.zrangebyscore(
                LEADERBOARD_KEY,
                boundary_score,
                boundary_score,
                withscores=True,
            )
            parsed: list[tuple[int, int]] = []
            for member, raw_score in raw_members:
                telegram_id = _parse_redis_member(member)
                if telegram_id is None:
                    malformed_members.append(member.decode() if isinstance(member, bytes) else str(member))
                    continue
                try:
                    score = coerce_leaderboard_score(raw_score)
                except Exception:
                    malformed_members.append(str(member))
                    continue
                parsed.append((telegram_id, score))

            if malformed_members:
                try:
                    client.zrem(LEADERBOARD_KEY, *malformed_members)
                except Exception:
                    logger.warning("Unable to clean malformed Telegram leaderboard members", exc_info=True)
                malformed_members = []

            if not parsed:
                continue

            profiles = {
                profile.telegram_id: profile
                for profile in TelegramProfile.objects.filter(telegram_id__in=[telegram_id for telegram_id, _ in parsed])
            }
            valid_group: list[tuple[int, int, TelegramProfile]] = []
            for telegram_id, score in parsed:
                profile = profiles.get(telegram_id)
                if profile is None or not profile.has_started_bot:
                    stale_members.append(leaderboard_member(telegram_id))
                    continue
                valid_group.append((telegram_id, score, profile))

            if stale_members:
                try:
                    client.zrem(LEADERBOARD_KEY, *stale_members)
                except Exception:
                    logger.warning("Unable to clean stale Telegram leaderboard members", exc_info=True)
                stale_members = []

            valid_group.sort(key=lambda item: item[0])
            rank = int(client.zcount(LEADERBOARD_KEY, f"({boundary_score}", "+inf")) + 1
            for telegram_id, score, profile in valid_group:
                rows.append({
                    "rank": rank,
                    "display_name": display_name(profile),
                    "score": score,
                })
                if len(rows) >= limit:
                    return rows[:limit]
        return rows[:limit]
    except Exception:
        logger.warning("Unable to fetch Telegram leaderboard rows", exc_info=True)
        raise
