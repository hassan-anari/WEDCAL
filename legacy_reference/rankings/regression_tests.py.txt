    def test_real_redis_rebuild_replaces_stale_and_replays_dirty_profiles(self):
        TelegramProfile.objects.create(telegram_id=1, has_started_bot=True, bonus_points=10)
        TelegramProfile.objects.create(telegram_id=2, has_started_bot=False, bonus_points=999)
        dirty = TelegramProfile.objects.create(telegram_id=3, has_started_bot=True, bonus_points=5)
        self.redis_client.zadd(leaderboard.LEADERBOARD_KEY, {"stale": 1000, "2": 999})

            leaderboard.sync_profile(dirty.pk, redis_client=client)

        with patch("apps.telegram_airdrop.leaderboard._build_tmp_leaderboard", side_effect=build_and_dirty):

        self.assertEqual(self.redis_client.zscore(leaderboard.LEADERBOARD_KEY, "1"), 10)
        self.assertEqual(self.redis_client.zscore(leaderboard.LEADERBOARD_KEY, "2"), None)
        self.assertEqual(self.redis_client.zscore(leaderboard.LEADERBOARD_KEY, "3"), 55)
        self.assertEqual(self.redis_client.exists(leaderboard.LEADERBOARD_REBUILD_LOCK_KEY), 0)
        self.assertEqual(self.redis_client.scard(leaderboard.LEADERBOARD_REBUILD_DIRTY_KEY), 0)

# --- RealRedisLeaderboardIntegrationTests.test_real_redis_lock_renewal_checks_token_and_extends_dirty_key_ttls (lines 437-461) ---
    def test_real_redis_lock_renewal_checks_token_and_extends_dirty_key_ttls(self):
        token = leaderboard.acquire_rebuild_lock(redis_client=self.redis_client, ttl=2)
        self.assertIsNotNone(token)
        self.redis_client.sadd(leaderboard.LEADERBOARD_REBUILD_DIRTY_KEY, "1")
        self.redis_client.expire(leaderboard.LEADERBOARD_REBUILD_DIRTY_KEY, 2)
        self.redis_client.sadd(leaderboard.LEADERBOARD_REBUILD_REMOVED_KEY, "2")
        self.redis_client.expire(leaderboard.LEADERBOARD_REBUILD_REMOVED_KEY, 2)

        before_lock_ttl = self.redis_client.ttl(leaderboard.LEADERBOARD_REBUILD_LOCK_KEY)
        before_dirty_ttl = self.redis_client.ttl(leaderboard.LEADERBOARD_REBUILD_DIRTY_KEY)
        before_removed_ttl = self.redis_client.ttl(leaderboard.LEADERBOARD_REBUILD_REMOVED_KEY)
        self.assertTrue(leaderboard.renew_rebuild_lock(token, redis_client=self.redis_client, ttl=20, dirty_ttl=20, removed_ttl=20))
        self.assertGreater(self.redis_client.ttl(leaderboard.LEADERBOARD_REBUILD_LOCK_KEY), before_lock_ttl)
        self.assertGreater(self.redis_client.ttl(leaderboard.LEADERBOARD_REBUILD_DIRTY_KEY), before_dirty_ttl)
        self.assertGreater(self.redis_client.ttl(leaderboard.LEADERBOARD_REBUILD_REMOVED_KEY), before_removed_ttl)

        after_lock_ttl = self.redis_client.ttl(leaderboard.LEADERBOARD_REBUILD_LOCK_KEY)
        after_dirty_ttl = self.redis_client.ttl(leaderboard.LEADERBOARD_REBUILD_DIRTY_KEY)
        after_removed_ttl = self.redis_client.ttl(leaderboard.LEADERBOARD_REBUILD_REMOVED_KEY)
        self.assertFalse(leaderboard.renew_rebuild_lock("wrong-token", redis_client=self.redis_client, ttl=60, dirty_ttl=60, removed_ttl=60))
        self.assertLessEqual(self.redis_client.ttl(leaderboard.LEADERBOARD_REBUILD_LOCK_KEY), after_lock_ttl)
        self.assertLessEqual(self.redis_client.ttl(leaderboard.LEADERBOARD_REBUILD_DIRTY_KEY), after_dirty_ttl)
        self.assertLessEqual(self.redis_client.ttl(leaderboard.LEADERBOARD_REBUILD_REMOVED_KEY), after_removed_ttl)
        self.redis_client.delete(leaderboard.LEADERBOARD_REBUILD_DIRTY_KEY, leaderboard.LEADERBOARD_REBUILD_REMOVED_KEY)
        self.assertTrue(leaderboard.release_rebuild_lock_if_owned_and_clean(token, redis_client=self.redis_client))
