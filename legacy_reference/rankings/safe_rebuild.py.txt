def _build_tmp_leaderboard(client, tmp_key: str, *, chunk_size: int, token: str) -> int:
    synced = 0
    pipe = client.pipeline()
    pending = 0
    queryset = TelegramProfile.objects.filter(has_started_bot=True).only(
        "id",
        "telegram_id",
        "activity_earned_points",
        "group_earned_points",
        "referral_earned_points",
        "bonus_points",
    ).order_by("id")
    for profile in queryset.iterator(chunk_size=chunk_size):
        pipe.zadd(tmp_key, {leaderboard_member(profile.telegram_id): current_score(profile)})
        synced += 1
        pending += 1
        if pending >= chunk_size:
            pipe.execute()
            ensure_rebuild_lock_owned(token, redis_client=client)
            pipe = client.pipeline()
            pending = 0
    if pending:
        pipe.execute()
        ensure_rebuild_lock_owned(token, redis_client=client)
    client.expire(tmp_key, REBUILD_TMP_TTL_SECONDS)
    return synced

def _drain_dirty_changes(client, *, max_rounds: int = 100) -> dict[str, int]:
    totals = {"profiles": 0, "removed": 0}
    for _ in range(max_rounds):
        dirty_ids = client.smembers(LEADERBOARD_REBUILD_DIRTY_KEY)
        removed_ids = client.smembers(LEADERBOARD_REBUILD_REMOVED_KEY)
        if not dirty_ids and not removed_ids:
            return totals
        if dirty_ids:
            client.srem(LEADERBOARD_REBUILD_DIRTY_KEY, *dirty_ids)
            profile_ids = []
            for value in dirty_ids:
                try:
                    profile_ids.append(int(value))
                except (TypeError, ValueError):
                    continue
            sync_profiles(profile_ids, redis_client=client, mark_dirty=False)
            totals["profiles"] += len(profile_ids)
        if removed_ids:
            client.srem(LEADERBOARD_REBUILD_REMOVED_KEY, *removed_ids)
            client.zrem(LEADERBOARD_KEY, *removed_ids)
            totals["removed"] += len(removed_ids)
    raise RuntimeError("Unable to drain leaderboard dirty changes before max rounds")

def rebuild_leaderboard(*, redis_client=None, chunk_size: int = 1000) -> dict[str, Any]:
    client = _client(redis_client)
    token = acquire_rebuild_lock(redis_client=client)
    if not token:
        raise RuntimeError("Telegram airdrop leaderboard rebuild is already running")

    tmp_key = f"{LEADERBOARD_REBUILD_TMP_PREFIX}{uuid.uuid4().hex}"
    started = time.monotonic()
    synced = 0
    dirty_replayed = 0
    removed_replayed = 0
    renamed = False
    try:
        client.delete(tmp_key)
        synced = _build_tmp_leaderboard(client, tmp_key, chunk_size=chunk_size, token=token)
        ensure_rebuild_lock_owned(token, redis_client=client)
        if synced == 0:
            client.delete(tmp_key)
            client.delete(LEADERBOARD_KEY)
        else:
            _promote_tmp_leaderboard(client, tmp_key)
            renamed = True
        drained = _drain_dirty_changes(client)
        dirty_replayed += drained["profiles"]
        removed_replayed += drained["removed"]
        while not release_rebuild_lock_if_owned_and_clean(token, redis_client=client):
            if client.get(LEADERBOARD_REBUILD_LOCK_KEY) != token:
                break
            drained = _drain_dirty_changes(client)
            dirty_replayed += drained["profiles"]
            removed_replayed += drained["removed"]
        return {
            "synced": synced,
            "dirty_replayed": dirty_replayed,
            "removed_replayed": removed_replayed,
            "renamed": renamed,
            "duration_seconds": round(time.monotonic() - started, 3),
        }
    except Exception:
        logger.warning("Telegram leaderboard rebuild failed", exc_info=True)
        raise
    finally:
        try:
            client.delete(tmp_key)
        except Exception:
            logger.warning("Unable to clean temporary Telegram leaderboard key", exc_info=True, extra={"tmp_key": tmp_key})
